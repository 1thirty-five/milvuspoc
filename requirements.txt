# Direct dependencies for the Milvus POC.
# Tested versions (2026-06-10): pymilvus 2.4.0, sentence-transformers 5.5.1
# (which pulls in torch 2.12.0, transformers 5.10.2, numpy 2.4.6).
#
# GPU NOTE: installing this file gives you the CPU build of torch, because that
# is what PyPI serves. Embedding is by far the slowest stage of the pipeline and
# it runs ~14x faster on a GPU (all 1072 chunks of fileinput/ with bge-m3: ~370s
# on CPU vs ~26s on an RTX 3050). If you have an NVIDIA card, install the CUDA
# build over the top -- same torch version, just compiled against CUDA 12.6:
#
#     pip install --index-url https://download.pytorch.org/whl/cu126 torch==2.12.0+cu126
#
# Confirm it took with:  python -c "import torch; print(torch.cuda.is_available())"
# The pipeline needs no config to use it: sentence-transformers picks the GPU up
# on its own, and every script prints which device the model landed on.
pymilvus>=2.4
sentence-transformers>=2.2
pypdfium2>=4.30     # PDF text layer + page rendering in extractpdf.py
pillow>=10.0        # image input and the render->OCR handoff
#
# OCR: baidu/Unlimited-OCR (https://github.com/baidu/Unlimited-OCR), a 3.34B
# vision-language model that parses a whole page in one shot into structured
# markdown. Replaces PyMuPDF's text-layer-only extraction, so scanned PDFs and
# images are ingestable instead of being skipped. MIT, fully local, no API key,
# no page cap. It is pulled from the HF hub at first use (~6.2 GiB), so there is
# no pip line for the model itself -- only the runtime it needs:
# PIN, do not float. Baidu pins transformers 4.57.1 and Unlimited-OCR's remote
# modeling code is written against it. On transformers 5.x the model loads but
# then fails in Baidu's custom sliding-window KV cache: that code tracks state as
# attributes on the cache object and branches on prefill-vs-decode, and v5 split
# prefill into its own method, so keys get appended twice while values are not.
# extractpdf.py shims the other v5 differences (all guarded by `hasattr`, so they
# are inert here), but the cache mismatch has no safe fix -- a wrong KV cache does
# not raise, it emits fluent, confident, WRONG text, which is the worst possible
# failure mode for OCR.
# 4.57.1 is safe for the rest of the pipeline: sentence-transformers 5.5.1 wants
# transformers>=4.41,<6, and bge-m3 was re-verified on it (1024-dim, normalized).
transformers==4.57.1
accelerate>=0.30    # device placement for the quantized load path
#
# Unlimited-OCR's remote modeling code imports these directly, and transformers
# refuses to load the module until every one is present, so they are hard
# requirements rather than extras:
addict>=2.4
easydict>=1.13
einops>=0.8
matplotlib>=3.8
#
# torchvision must match the installed torch. Do NOT `pip install torchvision`
# from PyPI here -- that resolves the CPU build and drags CPU torch in with it,
# undoing the CUDA install above. Take it from the same index as torch:
#
#     pip install --index-url https://download.pytorch.org/whl/cu126 torchvision==0.27.0+cu126
#
# (torch 2.12 pairs with torchvision 0.27. Verify with:
#  python -c "import torch,torchvision; print(torch.__version__, torchvision.__version__)")
torchvision         # version/build must match torch -- see the note above
#
# VRAM: the BF16 weights are 6.21 GiB and this machine's RTX 3050 has 6.0 GiB,
# so the default `--ocr-quant auto` loads 4-bit NF4 (~1.9 GiB) via bitsandbytes.
# That makes bitsandbytes effectively required for local OCR on this card:
#
#     pip install bitsandbytes
#
# Note the model's own code is hard-wired to CUDA (.cuda(), autocast("cuda")),
# so there is no CPU fallback. Without a suitable GPU, either serve the model
# elsewhere (--ocr-backend unlimited-server) or run --ocr never (text layer only,
# no GPU needed at all).
scikit-learn>=1.3   # KMeans clustering in cluster.py
umap-learn>=0.5     # 2D projection for visualize.py
plotly>=5.0         # interactive cluster plot in visualize.py
rank-bm25>=0.2      # BM25 lexical / hybrid retrieval in search.py
